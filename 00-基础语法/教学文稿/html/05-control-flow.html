05 循环与流程控制

学完本篇你将能够：用三种方式遍历数组/切片/Map、正确使用 break 与 continue、避免滥用 goto、写出带穿透的 switch。

📝 一篇一练🧩 小步掌握查看 Markdown 原稿 ↗
✎ 一句话记住

Go 的循环一把 for 梭；continue 跳过这轮，break 离开当前循环。

🧠 30 秒复述`switch` 默认会自动落到下一个 case 吗？`fallthrough` 呢？

默认不会；只有显式写 fallthrough 才会继续执行下一个 case。

概述

前几篇分享了 array 数组、slice 切片、map 集合，本篇讲如何循环获取里面的元素，同时也是对前几篇的复习。

涉及关键字：for、range、break、continue、goto、switch。

循环 array

GO
// s07_loop/main.go
package main

import "fmt"

func main() {
	person := [3]string{"Tom", "Aaron", "John"}
	fmt.Printf("len=%d cap=%d array=%v\n", len(person), cap(person), person)

	// 方式一：range 拿键值
	for k, v := range person {
		fmt.Printf("person[%d]: %s\n", k, v)
	}

	// 方式二：range 拿下标
	for i := range person {
		fmt.Printf("person[%d]: %s\n", i, person[i])
	}

	// 方式三：标准三段式
	for i := 0; i < len(person); i++ {
		fmt.Printf("person[%d]: %s\n", i, person[i])
	}

	// 只需要值时用空白符
	for _, name := range person {
		fmt.Println("name :", name)
	}
}

循环 slice

写法与数组完全一致，只是 cap 与 len 可能不同：

GO
person := []string{"Tom", "Aaron", "John"}
fmt.Printf("len=%d cap=%d slice=%v\n", len(person), cap(person), person)

for k, v := range person { /* ... */ }
for i := range person { /* ... */ }
for i := 0; i < len(person); i++ { /* ... */ }
for _, name := range person { /* ... */ }

⚠️ range 的值是副本。遍历切片时：

GO
for i, v := range s { v = 99 }        // s 不变——只改了副本 v
for i := range s { s[i] = 99 }        // s 被修改
for i, v := range s { s[i] = 99; _ = v } // 也可以，但写法更差

大对象的额外优化：不关心 value 时用 for i := range s，可避免每次迭代都拷贝元素（尤其是大 struct 或 string）。

循环 map

GO
person := map[int]string{
	1: "Tom",
	2: "Aaron",
	3: "John",
}
fmt.Printf("len=%d map=%v\n", len(person), person)

for k, v := range person {
	fmt.Printf("person[%d]: %s\n", k, v)
}

// 只需要 key
for i := range person {
	fmt.Printf("person[%d]: %s\n", i, person[i])
}

// 只需要 value
for _, name := range person {
	fmt.Println("name :", name)
}

⚠️ 不要写 for i := 1; i <= len(person); i++ { person[i] } —— Map 不保证 key 从 0 开始、连续或按顺序出现；需要遍历时使用 range。

⚠️ 遍历顺序随机（第 03 篇）。需要稳定顺序时先排序 key。

break

跳出当前循环，可用于 for、switch、select。

GO
func main() {
	for i := 1; i <= 10; i++ {
		if i == 6 {
			break
		}
		fmt.Println("i =", i)
	}
}

输出：1 2 3 4 5

在嵌套循环里 break 只跳出内层。要跳出多层用 break + 标签：

GO
outer:
for i := 0; i < 3; i++ {
	for j := 0; j < 3; j++ {
		if i == 1 && j == 1 {
			break outer // 跳出双层循环
		}
	}
}

continue

跳过本次循环，只能用于 for。

GO
func main() {
	for i := 1; i <= 10; i++ {
		if i == 6 {
			continue
		}
		fmt.Println("i =", i)
	}
}

输出：1 2 3 4 5 7 8 9 10

goto

改变函数内代码执行顺序，不能跨函数使用。

GO
func main() {
	fmt.Println("begin")

	for i := 1; i <= 10; i++ {
		if i == 6 {
			goto END
		}
		fmt.Println("i =", i)
	}

END:
	fmt.Println("end")
}

输出：

代码片段
begin
1 2 3 4 5
end

Go 没有 PHP 那样的 continue 2（多层 continue），多层跳出请用标签。

业务代码中几乎不应出现 goto。本篇仅用于理解底层跳转机制，Go 编译器本身在生成代码时会大量使用它（如 error 处理流程）。

switch

GO
func main() {
	for i := 1; i <= 7; i++ {
		fmt.Printf("当 i = %d 时：\n", i)
		switch i {
		case 1:
			fmt.Println("输出 i =", 1)
		case 2:
			fmt.Println("输出 i =", 2)
		case 3:
			fmt.Println("输出 i =", 3)
			fallthrough // 不跳出，继续执行下一个 case
		case 4, 5, 6: // 一个 case 可有多个选项
			fmt.Println("输出 i = 4 or 5 or 6")
		default:
			fmt.Println("输出 i = xxx")
		}
	}
}
i输出
1输出 i = 1
2输出 i = 2
3输出 i = 3，输出 i = 4 or 5 or 6
4 / 5 / 6输出 i = 4 or 5 or 6
7输出 i = xxx

结论：

  • 默认每个 case 带有 break（不像 C 需要手写 break）
  • case 中可以有多个选项，用 , 分隔
  • fallthrough 不跳出，并执行下一个 case（只能穿透一层，不能穿透多 case 到指定位置）
  • default 匹配所有未命中的情况

switch 的三种进阶用法

1. 无表达式 switch（替代 if...else if 链）

GO
score := 85
var grade string

switch {
case score >= 90:
	grade = "A"
case score >= 60:
	grade = "B"
default:
	grade = "C"
}
fmt.Println(grade) // B

相比 if...else if 链，Go 的 switch 不会「忘记写 break」导致贯穿，且支持任意布尔表达式（不限于相等比较）。

2. 类型 switch

GO
func describe(i interface{}) {
	switch v := i.(type) {
	case int:
		fmt.Println("int:", v)
	case string:
		fmt.Println("string:", len(v))
	case []int:
		fmt.Println("[]int 长度:", len(v))
	case error:
		fmt.Println("error:", v.Error())
	case nil:
		fmt.Println("nil")
	default:
		fmt.Println("未知类型")
	}
}

switch v := i.(type) 中的 v 在每个 case 内是该 case 对应的具体类型，这比 i.(int) 的类型断言实用得多。

⚠️ 若多个 case 都需要类型，用 switch i.(type)（不接收值），需要值时再在 case 内断言。

3. 标签穿透到 case 上

GO
for _, grade := range []string{"A", "B", "C"} {
	switch {
	case grade == "A":
		fmt.Println("优秀")
	case grade == "B":
		fmt.Println("良好")
	default:
		fmt.Println("及格")
	}
}

4. 匹配任意可比较类型

GO
var val interface{} = "go"
switch val.(type) {
case string, int: // 一个 case 列出多种类型
	fmt.Println("string 或 int")
}

本篇要点

  1. 遍历数组/切片：range k,v / range i / 三段式 for；遍历 Map 只能 range。
  2. range 的 value 是副本，改原容器要 s[i] = x。
  3. 不需要 value 时写 for i := range s，避免无意义的元素拷贝。
  4. break 跳出一层（多层用标签 break outer），continue 跳过一次，goto 改执行顺序（勿在业务代码用）。
  5. Go 的 switch 默认不穿透、支持多值 case、支持无表达式 switch 和类型 switch，不需要手写 break。
  6. for i := 0; i < n; i++ 与 for i := range n（Go 1.22+）在遍历整数时更简洁。

// s07_loop/main.go —— 对应教学文稿 05 循环与流程控制
//
// 运行：go run ./s07_loop
package main

import "fmt"

func main() {
	// ==================== 循环 array ====================
	person := [3]string{"Tom", "Aaron", "John"}
	fmt.Printf("len=%d cap=%d array=%v\n", len(person), cap(person), person)

	for k, v := range person { // 方式一：range 拿键值
		fmt.Printf("person[%d]: %s\n", k, v)
	}
	for i := range person { // 方式二：range 拿下标
		fmt.Printf("person[%d]: %s\n", i, person[i])
	}
	for i := 0; i < len(person); i++ { // 方式三：标准三段式
		fmt.Printf("person[%d]: %s\n", i, person[i])
	}
	for _, name := range person { // 空白符
		fmt.Println("name :", name)
	}

	// ==================== 循环 slice ====================
	fmt.Println("\n--- slice ---")
	sl := []string{"Tom", "Aaron", "John"}
	fmt.Printf("len=%d cap=%d slice=%v\n", len(sl), cap(sl), sl)

	for k, v := range sl {
		fmt.Printf("person[%d]: %s\n", k, v)
	}

	// range 的 value 是副本
	s := []int{1, 2, 3}
	for _, v := range s {
		v = 99 // 只改副本
		fmt.Println("副本 v =", v)
	}
	fmt.Println("改副本后：", s) // [1 2 3]

	for i := range s {
		s[i] = 99 // 改原切片
	}
	fmt.Println("改原切片：", s) // [99 99 99]

	// ==================== 循环 map ====================
	fmt.Println("\n--- map ---")
	personMap := map[int]string{
		1: "Tom",
		2: "Aaron",
		3: "John",
	}
	for k, v := range personMap {
		fmt.Printf("person[%d]: %s\n", k, v)
	}
	for i := range personMap {
		fmt.Printf("person[%d]: %s\n", i, personMap[i])
	}
	for _, name := range personMap {
		fmt.Println("name :", name)
	}

	// ==================== break ====================
	fmt.Println("\n--- break ---")
	for i := 1; i <= 10; i++ {
		if i == 6 {
			break // 跳出当前循环
		}
		fmt.Println("i =", i)
	}

	// break + 标签跳出多层
	fmt.Println("双层循环 break outer：")
outer:
	for i := 0; i < 3; i++ {
		for j := 0; j < 3; j++ {
			if i == 1 && j == 1 {
				fmt.Printf("跳出于 i=%d j=%d\n", i, j)
				break outer
			}
		}
	}

	// ==================== continue ====================
	fmt.Println("\n--- continue ---")
	for i := 1; i <= 10; i++ {
		if i == 6 {
			continue // 跳过本次
		}
		fmt.Println("i =", i)
	}

	// ==================== goto ====================
	fmt.Println("\n--- goto ---")
	fmt.Println("begin")
	for i := 1; i <= 10; i++ {
		if i == 6 {
			goto END
		}
		fmt.Println("i =", i)
	}
END:
	fmt.Println("end")

	// ==================== switch ====================
	fmt.Println("\n--- switch ---")
	for i := 1; i <= 7; i++ {
		fmt.Printf("当 i = %d 时：", i)
		switch i {
		case 1:
			fmt.Println("输出 i =", 1)
		case 2:
			fmt.Println("输出 i =", 2)
		case 3:
			fmt.Println("输出 i =", 3)
			fallthrough // 穿透一层
		case 4, 5, 6: // 多值 case
			fmt.Println("输出 i = 4 or 5 or 6")
		default:
			fmt.Println("输出 i = xxx")
		}
	}

	// 无表达式 switch
	fmt.Println("\n--- 无表达式 switch ---")
	score := 85
	var grade string
	switch {
	case score >= 90:
		grade = "A"
	case score >= 60:
		grade = "B"
	default:
		grade = "C"
	}
	fmt.Println("grade =", grade)

	// 类型 switch
	fmt.Println("\n--- 类型 switch ---")
	describe := func(i interface{}) {
		switch v := i.(type) {
		case int:
			fmt.Println("int:", v)
		case string:
			fmt.Println("string 长度:", len(v))
		case []int:
			fmt.Println("[]int 长度:", len(v))
		case error:
			fmt.Println("error:", v.Error())
		case nil:
			fmt.Println("nil")
		default:
			fmt.Println("未知类型")
		}
	}
	describe(1)
	describe("go")
	describe([]int{1, 2, 3})
	describe(nil)
	describe(3.14)
}
